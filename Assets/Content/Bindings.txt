# Commands and the keys bound to them (plan task 1.5; blueprint ch. 09 §3.1).
#
#   context name= priority=        a set of bindings live together; the higher priority wins a key
#                                  bound in two live contexts
#   command name= hold= repeat=    hold: released when its key goes up. repeat: takes auto-repeat
#   bind command= context= key=    key: a letter or digit, F1..F24, Tab, Space, Escape, Home, Up,
#                                  Down, Left, Right, MouseLeft, MouseRight, MouseMiddle, WheelUp,
#                                  WheelDown (NeuronClient/CommandMap.cpp has the full list)
#
# Always is live in every view; Flying in the chase view; Tactical in the tactical view. A key held
# while the view changes releases what it pressed, so a held thruster cannot stick on.
#
# The tactical view's mouse is the view's own, not a binding: left drag orbits and a left click selects,
# right or middle drag pans, a right click orders the own ship to go there, and the wheel zooms.

context name=Always priority=0
context name=Flying priority=10
context name=Tactical priority=10

# Flight: throttle and engine.
command name=ThrottleUp hold=true
command name=ThrottleDown hold=true
command name=ThrottleStepUp
command name=ThrottleStepDown
command name=Reverse hold=true
command name=EngineKill
command name=Cruise
command name=Thruster hold=true

# Flight: strafe and keyboard steering.

command name=StrafeRight hold=true
command name=StrafeUp hold=true
command name=StrafeDown hold=true
command name=RollLeft hold=true
command name=RollRight hold=true

command name=NoseDown hold=true
command name=YawLeft hold=true
command name=YawRight hold=true

# Flight: the mouse. Hold the left button to steer by the cursor, or lock it on with Space.
command name=MouseFlight hold=true
command name=MouseFlightLock

# Flight: combat (plan tasks 3.4 and 3.11; GameDesign.md D22). Hold the right button to fire at the
# point under the cursor; a repair kit patches the hull in the field.
command name=Fire hold=true
command name=RepairKit

# Tactical.
command name=ResetView

# Docking (plan tasks 2.3 and 2.4): Dock asks the nearest station within 10 km; Cancel leaves the
# approach or the waiting list; Launch leaves the berth.
command name=Dock
command name=Launch

command name=Help
