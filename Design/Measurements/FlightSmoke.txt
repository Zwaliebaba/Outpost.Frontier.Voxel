# The flight smoke run (plan task 1.11): the player's escort, E-114, flown by its commands and then
# ordered from the tactical view. Pixels of the 1920 x 1080 smoke frame; ticks are server ticks, one
# per frame. Keys are named as in Assets/Content/Bindings.txt.
#
# Full throttle over a second, and at once a hard turn away from the frigate parked ahead: the
# cursor at the right edge, the left button held past the quarter-second that arms it.
key   tick=1   name=W ticks=60
move  tick=9   at=1900,500
key   tick=10  name=MouseLeft ticks=75
# The thruster for a second, then the cursor back to the centre.
key   tick=120 name=Tab ticks=60
move  tick=190 at=960,540
# The throttle back to nothing, so that what moves the ship after is the order alone. Then the
# tactical view, and a go-to order on a point by a right click; the ship flies it until the end.
key   tick=230 name=S ticks=60
press tick=300 name=F9
click tick=310 at=760,640 button=right
move  tick=480 at=960,540
